static shouldDocumentAllowWebAudioToAutoPlay(const Document&):
    if (document.isCapturing()) return true;
    if (document.quirks().shouldAutoplayWebAudioForArbitraryUserGesture() && mainDocument && mainDocument->hasHadUserInteraction()) return true;   // site-specific quirk list only
    return window && window->hasTransientActivation();
